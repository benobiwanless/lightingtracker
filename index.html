// Initialize the Leaflet Map centered over default coordinates const map = L.map('map').setView([28.024, -81.794], 8); // Load a dark theme tile layer to simulate night sky/storm tracking environment L.tileLayer('https://{s}://{z}/{x}/{y}{r}.png', { attribution: '© OpenStreetMap contributors © CARTO' }).addTo(map); // Layer group to hold the active lightning markers let lightningLayer = L.layerGroup().addTo(map); async function fetchLightningData() { const key = document.getElementById('apiKey').value.trim(); const lat = document.getElementById('lat').value; const lon = document.getElementById('lon').value; const status = document.getElementById('statusMsg'); if (!key) { status.innerText = "⚠️ Missing API Key!"; status.style.color = "#f85149"; return; } status.innerText = "Scanning skies for lightning activity..."; status.style.color = "#58a6ff"; lightningLayer.clearLayers(); // Use current unix timestamps for OpenWeather severe weather queries const end = Math.floor(Date.now() / 1000); const start = end - (3600 * 2); // Pull past 2 hours of records // Correct OpenWeather API structure for localized weather/severe alerts data const url = `https://openweathermap.org{lat}&lon=${lon}&appid=${key}`; try { const response = await fetch(url); if (!response.ok) throw new Error(`HTTP Error: ${response.status}`); const data = await response.json(); // Parse the alerts array looking specifically for convective/lightning events if (data && data.alerts && data.alerts.length > 0) { let strikeCount = 0; data.alerts.forEach(alert => { // OpenWeather returns localized weather alerts; we extract coordinates if available if(alert.event.toLowerCase().includes('thunderstorm') || alert.event.toLowerCase().includes('lightning')) { strikeCount++; // Fallback to searching area center if precise stroke coordinates aren't arrayed const markerLat = alert.lat || lat; const markerLon = alert.lon || lon; const marker = L.circleMarker([markerLat, markerLon], { radius: 12, fillColor: '#ffea00', color: '#ff9100', weight: 2, opacity: 1, fillOpacity: 0.8 }); marker.bindPopup(`${alert.event}
${alert.description}`); lightningLayer.addLayer(marker); } }); if (strikeCount > 0) { status.innerText = `Success! Plotted active storm systems in your zone.`; status.style.color = "#3fb950"; } else { status.innerText = "No severe lightning parameters matched in current active alerts."; status.style.color = "#e3b341"; } } else { status.innerText = "No active storm alerts detected in this range."; status.style.color = "#e3b341"; } } catch (error) { console.error(error); status.innerText = "Error pulling data. Verify subscription tier permissions or API Key."; status.style.color = "#f85149"; } }
